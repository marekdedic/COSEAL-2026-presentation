\begin{frame}{The problem}
	GNNs are highly sensitive to hyperparameter choices -- and graph data is very diverse.

	\bigskip

	\textbf{Q:} How to optimize hyperparameters for GNNs? Which HPO method should I use?

	\bigskip

	\uncover<2->{\textbf{A:} Nobody knows.}
\end{frame}

\begin{frame}{The benchmark}
	\begin{columns}[t]
		\begin{column}{0.48\textwidth}
			\textbf{HPO methods:}
			\begin{itemize}
				\item Grid Search
				\item Random Search
				\item Bayesian Optimization (BO)
				\item Tree-structured Parzen Estimator (TPE)
				\item Sobol Quasi-Monte Carlo (QMC)
			\end{itemize}
		\end{column}
		\begin{column}{0.48\textwidth}
			\textbf{Graph datasets:} \\
			Cora, CiteSeer, PubMed, Squirrel, DBLP, CoraFull, Computers, Flickr, OGB ArXiv \\
			(2.7k -- 169k nodes)
		\end{column}
	\end{columns}

	\bigskip
	10 independent runs per method per dataset.
\end{frame}
